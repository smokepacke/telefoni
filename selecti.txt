CREATE VIEW user_purchase_statistics AS
SELECT
    u.id_user,
    u.login_user,
    u.fam_user,
    COUNT(p.id_purchase) AS purchases_count,
    COALESCE(SUM(p.amount_purchase), 0) AS total_amount
FROM users u
LEFT JOIN purchases p ON u.id_user = p.id_user
GROUP BY u.id_user, u.login_user, u.fam_user;


SELECT *
FROM user_purchase_statistics
ORDER BY total_amount DESC;

WITH user_totals AS (
    SELECT
        id_user,
        SUM(amount_purchase) AS total_amount
    FROM purchases
    GROUP BY id_user
)
SELECT
    u.login_user,
    u.fam_user,
    ut.total_amount
FROM users u
JOIN user_totals ut ON u.id_user = ut.id_user
WHERE ut.total_amount > 150000;
